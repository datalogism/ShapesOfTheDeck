```shacl
@prefix dbo: <http://dbpedia.org/ontology/> .
@prefix foaf: <http://xmlns.com/foaf/0.1/> .
@prefix rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#> .

@prefix sh: <http://www.w3.org/ns/shacl#> .
@prefix xsd: <http://www.w3.org/2001/XMLSchema#> .

shapes:AstronautShape a sh:NodeShape ;
    sh:targetClass dbo:Astronaut ;
    sh:property [
        sh:path rdfs:label ;
        sh:minCount 1 ;
        sh:nodeKind sh:Literal ;
        sh:datatype rdf:langString
    ], [
        sh:path foaf:name ;
        sh:minCount 1 ;
        sh:nodeKind sh:Literal ;
        sh:datatype xsd:string
    ], [
        sh:path dbo:wikiPageRevisionID ;
        sh:minCount 1 ;
        sh:nodeKind sh:Literal ;
        sh:datatype xsd:integer
    ], [
        sh:path rdfs:comment ;
        sh:minCount 1 ;
        sh:nodeKind sh:Literal ;
        sh:datatype rdf:langString
    ], [
        sh:path dbo:birthDate ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:nodeKind sh:Literal ;
        sh:datatype xsd:date
    ], [
        sh:path dbo:birthYear ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
        sh:nodeKind sh:Literal ;
        sh:datatype xsd:gYear
    ], [
        sh:path dbo:birthPlace ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class dbo:Place
    ], [
        sh:path dbo:nationality ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class dbo:Country
    ], [
        sh:path dbo:occupation ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI ;
        sh:class dbo:Profession
    ], [
        sh:path dbo:status ;
        sh:minCount 1 ;
        sh:nodeKind sh:Literal ;
        sh:datatype xsd:string ;
        sh:in ("Active" "Retired" "Deceased")
    ], [
        sh:path dbo:selection ;
        sh:minCount 1 ;
        sh:nodeKind sh:IRI
    ], [
        sh:path dbo:mission ;
        sh:nodeKind sh:IRI ;
        sh:class dbo:SpaceMission
    ], [
        sh:path dbo:almaMater ;
        sh:nodeKind sh:IRI ;
        sh:class dbo:EducationalInstitution
    ], [
        sh:path dbo:timeInSpace ;
        sh:nodeKind sh:Literal ;
        sh:datatype xsd:double
    ], [
        sh:path dbo:deathDate ;
        sh:maxCount 1 ;
        sh:nodeKind sh:Literal ;
        sh:datatype xsd:date
    ], [
        sh:path dbo:deathYear ;
        sh:maxCount 1 ;
        sh:nodeKind sh:Literal ;
        sh:datatype xsd:gYear
    ], [
        sh:path dbo:deathPlace ;
        sh:nodeKind sh:IRI ;
        sh:class dbo:Place
    ], [
        sh:path dbo:birthName ;
        sh:nodeKind sh:Literal ;
        sh:datatype xsd:string

        sh:path dbo:award ;
        sh:nodeKind sh:IRI ;
        sh:class dbo:Award
    ] .
```
